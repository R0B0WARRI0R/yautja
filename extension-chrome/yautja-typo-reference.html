6. Usage

// Inject HUD into any page:
yautja_act({
  type: 'evaluate',
  expression: require('fs').readFileSync('./extension/yautja-hud.js', 'utf8')
});
// Or inject via Helmet:
Observe: "Inject Yautja HUD on this page"
// Remove HUD:
yautja_act({
  type: 'evaluate',
  expression: `document.getElementById('yautja-hud')?.remove(); delete window.__yautjaHudLoaded`
});
